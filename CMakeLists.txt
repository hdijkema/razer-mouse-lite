function(embed_string name contents output)
    string(LENGTH "${contents}" size)
    set(code "inline const std::string ${name} = [] { std::string result; result.reserve(${size});\n")
    set(offset 0)
    while(offset LESS size)
        math(EXPR remaining "${size} - ${offset}")
        if(remaining GREATER 12000)
            string(SUBSTRING "${contents}" ${offset} 12000 chunk)
            string(FIND "${chunk}" "\n" end REVERSE)
            if(end GREATER_EQUAL 0)
                math(EXPR length "${end} + 1")
            else()
                set(length 12000)
            endif()
        else()
            set(length ${remaining})
        endif()
        string(SUBSTRING "${contents}" ${offset} ${length} chunk)
        string(APPEND code "result.append(R\"RML_EMBED(${chunk})RML_EMBED\");\n")
        math(EXPR offset "${offset} + ${length}")
    endwhile()
    string(APPEND code "return result; }();")
    set(${output} "${code}" PARENT_SCOPE)
endfunction()

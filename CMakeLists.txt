cmake_minimum_required(VERSION 3.21)
project(razer-mouse VERSION 0.3.0 LANGUAGES CXX)
set(MOUSE_LITE_VERSION "${PROJECT_VERSION}" CACHE STRING "Release version (X.Y.Z)")
if(NOT MOUSE_LITE_VERSION MATCHES "^(0|[1-9][0-9]*)[.](0|[1-9][0-9]*)[.](0|[1-9][0-9]*)$")
    message(FATAL_ERROR "MOUSE_LITE_VERSION must be X.Y.Z without leading zeroes")
endif()
string(REPLACE "." ";" version_parts "${MOUSE_LITE_VERSION}")
foreach(part IN LISTS version_parts)
    if(part GREATER 65535)
        message(FATAL_ERROR "Version components must fit Windows version resources (0..65535)")
    endif()
endforeach()
string(REPLACE "." "," MOUSE_LITE_VERSION_COMMAS "${MOUSE_LITE_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent LinguistTools)

add_subdirectory(src/core)
add_subdirectory(src/platform)
add_subdirectory(src/device)
add_subdirectory(src/ui)

qt_add_executable(razer-mouse WIN32 src/app/main.cpp tests/selftest.cpp resources.qrc)
if(WIN32)
    configure_file(src/app/windows.rc.in "${CMAKE_CURRENT_BINARY_DIR}/windows.rc" @ONLY)
    target_sources(razer-mouse PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/windows.rc")
endif()
target_include_directories(razer-mouse PRIVATE tests)
target_compile_definitions(razer-mouse PRIVATE MOUSE_LITE_VERSION="${MOUSE_LITE_VERSION}")
target_link_libraries(razer-mouse PRIVATE mouse_ui Qt6::Network)
set(TRANSLATION_FILES)
foreach(language en nl de da nb sv fr es)
    list(APPEND TRANSLATION_FILES "${CMAKE_CURRENT_SOURCE_DIR}/translations/mouse-lite_${language}.ts")
endforeach()
file(GLOB_RECURSE TRANSLATION_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/*.h")
qt_add_translations(razer-mouse TS_FILES ${TRANSLATION_FILES} SOURCES ${TRANSLATION_SOURCES}
    RESOURCE_PREFIX "/i18n" LUPDATE_OPTIONS -no-obsolete LRELEASE_OPTIONS -nounfinished)

foreach(target mouse_core mouse_platform mouse_device mouse_ui razer-mouse)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8 /MP)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME module-boundaries COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_CURRENT_SOURCE_DIR}" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/check-boundaries.cmake")
    if(WIN32)
        set(TEST_QPA_PLATFORM windows)
    else()
        set(TEST_QPA_PLATFORM offscreen)
    endif()
    add_test(NAME core-and-platform-contracts COMMAND razer-mouse --self-test --language en --catalogues "${CMAKE_CURRENT_SOURCE_DIR}/translations")
    set_tests_properties(core-and-platform-contracts PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${TEST_QPA_PLATFORM}")
    foreach(language en nl de da nb sv fr es)
        add_test(NAME ui-${language} COMMAND razer-mouse --smoke-test "${CMAKE_CURRENT_BINARY_DIR}/preview-${language}.png" --language ${language})
        set_tests_properties(ui-${language} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=${TEST_QPA_PLATFORM}")
    endforeach()
endif()
include(GNUInstallDirs)
install(TARGETS razer-mouse RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES LICENSE THIRD_PARTY.md README.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/razer-mouse-lite)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(FILES assets/razer-mouse-lite.png DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/256x256/apps)
    install(FILES packaging/linux/mouse-lite.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES packaging/linux/70-mouse-lite.rules DESTINATION ${CMAKE_INSTALL_DATADIR}/mouse-lite)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(CPACK_GENERATOR DEB)
    set(CPACK_PACKAGE_NAME razer-mouse-lite)
    set(CPACK_PACKAGE_VERSION "${MOUSE_LITE_VERSION}")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Compact Razer mouse profiles, DPI and lighting")
    set(CPACK_DEBIAN_PACKAGE_MAINTAINER "Razer Mouse Lite contributors")
    set(CPACK_DEBIAN_PACKAGE_SECTION utils)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_PACKAGE_FILE_NAME "razer-mouse-lite-${MOUSE_LITE_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
    include(CPack)
endif()

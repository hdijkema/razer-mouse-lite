if(WIN32)
    add_library(mouse_platform STATIC windows/windowsplatform.cpp windows/desktopintegration.cpp desktopentry.cpp)
    target_compile_definitions(mouse_platform PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(mouse_platform PRIVATE hid setupapi user32 shell32 advapi32 winhttp gdi32)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(UDEV REQUIRED IMPORTED_TARGET libudev)
    find_package(CURL REQUIRED)
    add_library(mouse_platform STATIC linux/linuxplatform.cpp linux/desktopintegration.cpp desktopentry.cpp)
    target_link_libraries(mouse_platform PRIVATE PkgConfig::UDEV CURL::libcurl ${CMAKE_DL_LIBS})
else()
    message(FATAL_ERROR "Supported platforms: Windows and Linux. Add a platform implementation for this OS.")
endif()
target_sources(mouse_platform PRIVATE trayicon.cpp)
target_include_directories(mouse_platform PRIVATE "${PROJECT_BINARY_DIR}/generated")
target_link_libraries(mouse_platform PRIVATE fltk::images)
target_include_directories(mouse_platform PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(mouse_platform PUBLIC mouse_core)
